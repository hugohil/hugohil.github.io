Variation

map

map() convertit une valeur d'un intervalle vers un autre.

map(valeur, fromMin, fromMax, toMin, toMax)
background(10);

noStroke();
for (let i = 0; i < 10; i++) {
  let size = map(i, 0, 10, 10, 200);
  let a = map(i, 0, 10, 255, 0); // décroissant
  fill(255, a);
  circle(width/2, height / 2, size);
}

Contraindre

map() peut dépasser hors des bornes si la valeur les dépasse. constrain() permet d'éviter ça.

let size = constrain(map(mouseX, 0, width/2, 10, 200), 10, 200);

Il existe un cinquième argument optionnel : map(v, a, b, c, d, true) contraint directement le résultat.

dist

let cx = width / 2;
let cy = height / 2;

let steps = 20;
for (let x = 0; x < width; x += steps) {
  for (let y = 0; y < height; y += steps) {
    let d = dist(x, y, cx, cy);   // distance au centre
    let size = map(d, 0, 300, 18, 2);
    circle(x, y, size);
  }
}

Facilitez vous l'experimentation

// difficile d'experimenter
for (let i = 0; i < 24; i++) {
  circle(i * 25 + 12, 300, 18);
}

// parametrable
let total = 24;
let size = 18;
let h = 300;

for (let i = 0; i < total; i++) {
  let span = width / total;
  circle(i * span + span / 2, h, size);
}
